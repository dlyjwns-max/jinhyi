2. Unity C# 핵심 사격 스크립트

using UnityEngine;

public class AimAndShootingController : MonoBehaviour
{
    [Header("Aiming Angles")]
    public float yaw = 0f;
    public float pitch = 0f;
    public float sensitivity = 0.15f;

    [Header("Recoil Recovery")]
    public Vector2 recoilOffset = Vector2.zero;

    void Update()
    {
        // Smooth camera rotation combining player aim and recoil impulse
        pitch = Mathf.Clamp(pitch, -80f, 80f);
        recoilOffset = Vector2.Lerp(recoilOffset, Vector2.zero, Time.deltaTime * 10f);

        transform.localRotation = Quaternion.Euler(pitch + recoilOffset.y, yaw + recoilOffset.x, 0f);
    }
}
